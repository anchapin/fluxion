    pub fn value_for_day(&self, day_type: DayType, hour: usize) -> f64 {
        let day_idx = match day_type {
            DayType::Weekday => 0,
            DayType::Weekend => 5,
            DayType::Holiday => 0,
            DayType::Monday => 0,
            DayType::Tuesday => 1,
            DayType::Wednesday => 2,
            DayType::Thursday => 3,
            DayType::Friday => 4,
            DayType::Saturday => 5,
            DayType::Sunday => 6,
        };
        match &self.values {
            ScheduleValues::Daily(arr) => arr[hour % 24],
            ScheduleValues::Weekly(weekly) => weekly[day_idx][hour % 24],
        }
    }
}

impl DailySchedule {
    pub fn office_hours(mut self) -> Self {
        if let ScheduleValues::Weekly(ref mut weekly) = self.values {
            for day in 0..5 {  // Monday-Friday
                for hour in 8..=17 {  // 8am-6pm
                    weekly[day][hour] = 1.0;
                }
            }
        }
        self
    }

    pub fn fill_weekday(&mut self, start_hour: usize, end_hour: usize, value: f64) {
        if let ScheduleValues::Weekly(ref mut weekly) = self.values {
            for day in 0..5 {
                for hour in start_hour..end_hour {
                    weekly[day][hour] = value;
                }
            }
        }
    }

    pub fn fill_weekend(&mut self, start_hour: usize, end_hour: usize, value: f64) {
        if let ScheduleValues::Weekly(ref mut weekly) = self.values {
            for day in 5..7 {
                for hour in start_hour..end_hour {
                    weekly[day][hour] = value;
                }
            }
        }
    }

    pub fn fill_holiday(&mut self, start_hour: usize, end_hour: usize, value: f64) {
        if let ScheduleValues::Weekly(ref mut weekly) = self.values {
            for day in 0..7 {
                for hour in start_hour..end_hour {
                    weekly[day][hour] = value;
                }
            }
        }
    }
}

use std::collections::HashSet;
use std::sync::OnceLock;

use crate::sim::schedule::DayType;

static US_FEDERAL_HOLIDAYS: OnceLock<HashSet<usize>> = OnceLock::new();

/// Calculate US federal holidays for a given year
/// Returns HashSet of day_of_year (1-366)
fn calculate_holidays_for_year() -> HashSet<usize> {
    let mut holidays = HashSet::new();

    // Year-agnostic formulas for US federal holidays
    // New Year's Day (Jan 1)
    holidays.insert(1);

    // Martin Luther King Jr Day (3rd Monday in January)
    // Jan 1-7: check for Monday, Jan 8-14: check for Monday, etc.
    for day in 1..31 {
        let day_of_week = (day - 1) % 7; // 0=Monday
        if day_of_week == 0 && (1..=7).contains(&day) && is_nth_monday(day, 3) {
            holidays.insert(day);
        }
    }

    // Presidents' Day (3rd Monday in February)
    for day in 32..60 {
        let day_of_week = (day - 32) % 7;
        if day_of_week == 0 && is_nth_monday(day - 31, 3) {
            holidays.insert(day);
        }
    }

    // Memorial Day (last Monday in May)
    for day in (122..152).rev() {
        let day_of_week = (day - 122) % 7;
        if day_of_week == 0 {
            holidays.insert(day);
            break;
        }
    }

    // Juneteenth (June 19)
    holidays.insert(170);

    // Independence Day (July 4)
    holidays.insert(185);

    // Labor Day (1st Monday in September)
    for day in 244..270 {
        let day_of_week = (day - 244) % 7;
        if day_of_week == 0 && is_nth_monday(day - 243, 1) {
            holidays.insert(day);
        }
    }

    // Columbus Day (2nd Monday in October)
    for day in 274..304 {
        let day_of_week = (day - 274) % 7;
        if day_of_week == 0 && is_nth_monday(day - 273, 2) {
            holidays.insert(day);
        }
    }

    // Veterans Day (November 11)
    holidays.insert(315);

    // Thanksgiving (4th Thursday in November)
    for day in 306..336 {
        let day_of_week = (day - 306) % 7;
        if day_of_week == 3 && is_nth_thursday(day - 305, 4) {
            holidays.insert(day);
        }
    }

    // Christmas (December 25)
    holidays.insert(359);

    holidays
}

fn is_nth_monday(day: usize, n: usize) -> bool {
    ((day - 1) / 7 + 1) == n
}

fn is_nth_thursday(day: usize, n: usize) -> bool {
    ((day - 1) / 7 + 1) == n
}

/// Get day type for a given day of year
pub fn get_day_type(day_of_year: usize) -> DayType {
    let day_of_week = (day_of_year - 1) % 7; // 0=Monday, ..., 6=Sunday

    // Check if holiday
    let holidays = US_FEDERAL_HOLIDAYS.get_or_init(|| calculate_holidays_for_year());
    if holidays.contains(&day_of_year) {
        return DayType::Holiday;
    }

    // Weekday or Weekend
    match day_of_week {
        0..=4 => DayType::Weekday,  // Monday-Friday
        5..=6 => DayType::Weekend,  // Saturday-Sunday
        _ => DayType::Weekday,
    }
}
