fn test_8r3c_case_920() {

    // Reference: 4.3-5.2 MWh (EnergyPlus, ESP-r, TRNSYS)
    // 5R1C baseline: 14.1 MWh (229% error)

    // Verify CaseBuilder::case_920 exists (from Phase 18 diagnostic cases)

    let weather = DenverTmyWeather::new();
    let weather_data = weather.get_annual_data();

    // Run 1-year simulation
    let energy = model.solve_timesteps_8r3c(8760, weather_data, false);

    // 8R3C should be within ±15% of reference range (ASHRAE 140 tolerance)

    let reference_max = 5.2;
