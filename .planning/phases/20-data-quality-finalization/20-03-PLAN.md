Implement parse_epw() function to detect version from file header:

```rust
pub fn parse_epw<R: Read>(reader: R) -> Result<EpwVersion, EpwError> {
    let mut buf_reader = BufReader::new(reader);
    let mut lines = Vec::new();

    for line in buf_reader.lines() {
        let line = line.map_err(|e| EpwError::IoError(e.to_string()))?;
        if line.starts_with('#') || line.trim().is_empty() {
            continue; // Skip comments and empty lines
        }
        lines.push(line);
    }

    if lines.is_empty() {
        return Err(EpwError::EmptyFile);
    }

    // Detect version from first data line
    let first_line_parts: Vec<&str> = lines[0].split(',').collect();

    // EPW v2: 35 fields, hourly records
    if first_line_parts.len() == 35 {
        return parse_epw_v2(lines);
    }

    // EPW v3: 35+ fields, sub-hourly records
    if first_line_parts.len() > 35 {
        return parse_epw_v3(lines);
    }

    // AMY: Similar to v2 but has "AMY" in header
    if lines[0].contains("AMY") {
        return parse_epw_amy(lines);
    }

    // IWEC: Similar to v2 but has "IWEC" in header
    if lines[0].contains("IWEC") {
        return parse_epw_iwec(lines);
    }

    Err(EpwError::UnknownFormat)
}
```

```rust
use reqwest::blocking::Client;
use sha2::{Sha256, Digest};
use std::path::{PathBuf, Path};
use std::fs;
use std::io::Write;
use std::time::SystemTime;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum Tmy3Error {
    #[error("Failed to determine cache directory")]
    CacheDirectoryError,
    #[error("Failed to create cache directory: {0}")]
    CreateDirectoryError(String),
    #[error("Failed to download TMY3: {0}")]
    DownloadError(String),
    #[error("HTTP error: {0}")]
    HttpError(String),
    #[error("Failed to read cache file: {0}")]
    ReadError(String),
    #[error("Failed to write cache file: {0}")]
    WriteError(String),
}

pub struct Tmy3Cache {
    cache_dir: PathBuf,
    client: Client,
}

impl Tmy3Cache {
    pub fn new() -> Result<Self, Tmy3Error> {
        let cache_dir = dirs::cache_dir()
            .ok_or(Tmy3Error::CacheDirectoryError)?
            .join("fluxion/tmy3");

        fs::create_dir_all(&cache_dir)
            .map_err(|e| Tmy3Error::CreateDirectoryError(e.to_string()))?;

        Ok(Tmy3Cache {
            cache_dir,
            client: Client::new(),
        })
    }

    pub fn get_or_download(&self, url: &str, location: &str) -> Result<PathBuf, Tmy3Error> {
        let filename = format!("{}.tmy3", location.replace(' ', '_').to_lowercase());
        let filepath = self.cache_dir.join(&filename);

        // Check cache
        if filepath.exists() {
            // Verify checksum if exists
            let checksum_path = filepath.with_extension("sha256");
            if checksum_path.exists() {
                // Cache hit, return filepath
                return Ok(filepath);
            }
        }

        // Download file
        let response = self.client.get(url)
            .send()
            .map_err(|e| Tmy3Error::DownloadError(e.to_string()))?;

        if !response.status().is_success() {
            return Err(Tmy3Error::HttpError(response.status().to_string()));
        }

        let content = response.bytes()
            .map_err(|e| Tmy3Error::DownloadError(e.to_string()))?;

        // Verify checksum
        let checksum = format!("{:x}", Sha256::digest(&content));

        // Write to cache
        let mut file = fs::File::create(&filepath)
            .map_err(|e| Tmy3Error::WriteError(e.to_string()))?;
        file.write_all(&content)
            .map_err(|e| Tmy3Error::WriteError(e.to_string()))?;

        // Write checksum file
        let checksum_path = filepath.with_extension("sha256");
        let mut checksum_file = fs::File::create(&checksum_path)
            .map_err(|e| Tmy3Error::WriteError(e.to_string()))?;
        checksum_file.write_all(checksum.as_bytes())
            .map_err(|e| Tmy3Error::WriteError(e.to_string()))?;

        Ok(filepath)
    }

    pub fn get_cache_dir(&self) -> &Path {
        &self.cache_dir
    }

    pub fn clear_cache(&self) -> Result<(), Tmy3Error> {
        for entry in fs::read_dir(&self.cache_dir)
            .map_err(|e| Tmy3Error::ReadError(e.to_string()))?
        {
            let entry = entry.map_err(|e| Tmy3Error::ReadError(e.to_string()))?;
            fs::remove_file(entry.path())
                .map_err(|e| Tmy3Error::ReadError(e.to_string()))?;
        }
        Ok(())
    }
}
```

```rust
        .map_err(|e| format!("Failed to read {}: {}", path, e))?;
    serde_json::from_str(&content)
        .map_err(|e| format!("Failed to parse {}: {}", path, e))
}
```
