pub struct EnsembleSurrogateManager {
    models: Vec<SurrogateManager>,
}

impl EnsembleSurrogateManager {
    pub fn new(models: Vec<SurrogateManager>) -> Self {
        Self { models }
    }

    pub fn predict_ensemble(&self, temps: &[f64]) -> Vec<f64> {
        // Real ensemble inference using ONNX Runtime
        let predictions: Vec<Vec<f64>> = self.models
            .iter()
            .map(|m| m.predict_loads(temps))
            .collect();

        // Average ensemble
        let n = predictions.len();
        let ensemble_pred = (0..predictions[0].len())
            .map(|i| predictions.iter().map(|p| p[i]).sum::<f64>() / n as f64)
            .collect();

        ensemble_pred
    }
}

}
