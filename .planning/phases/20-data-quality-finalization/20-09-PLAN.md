```rust
pub fn new_default_assembly(num_zones: usize, assembly_name: &str) -> Result<Self, String> {
    // Load assembly library
    let library = AssemblyLibrary::new("data/assemblies.yaml", "data/materials.yaml")
        .map_err(|e| format!("Failed to load assembly library: {}", e))?;

    // Get assembly by name
    let assembly = library.get_assembly(assembly_name)
        .ok_or_else(|| format!("Assembly '{}' not found", assembly_name))?;

    // Create ThermalModel with assembly
    Ok(Self::new_with_assembly(num_zones, assembly))
}
```
Constructor loads assembly from YAML configuration, validates it exists, and delegates to new_with_assembly(). Provides user-friendly API for loading predefined assemblies.
