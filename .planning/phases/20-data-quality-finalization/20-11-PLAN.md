pub fn new_with_validation(
    num_zones: usize,
    window_u_value: f64,
    hvac_setpoint: f64,
    h_tr_em: f64,
    h_tr_ms: f64,
    h_tr_is: f64,
    h_tr_w: f64,
    h_ve: f64,
) -> Result<Self, String> {
    // Validate constants module
    let constants_result = validate_constants();
    if constants_result.validation != "passed" {
        let errors: Vec<String> = constants_result.errors
            .into_iter()
            .map(|e| format!("{}: {}", e.field, e.message))
            .collect();
        return Err(format!("Constants validation failed: {}", errors.join("; ")));
    }

    // Validate thermal conductances
    if h_tr_em <= 0.0 {
        return Err(format!("Invalid h_tr_em: {} (must be positive)", h_tr_em));
    }
    if h_tr_ms <= 0.0 {
        return Err(format!("Invalid h_tr_ms: {} (must be positive)", h_tr_ms));
    }
    if h_tr_is <= 0.0 {
        return Err(format!("Invalid h_tr_is: {} (must be positive)", h_tr_is));
    }
    if h_tr_w <= 0.0 {
        return Err(format!("Invalid h_tr_w: {} (must be positive)", h_tr_w));
    }
    if h_ve <= 0.0 {
        return Err(format!("Invalid h_ve: {} (must be positive)", h_ve));
    }

    // Validate HVAC setpoint
    if hvac_setpoint < 15.0 || hvac_setpoint > 30.0 {
        return Err(format!("Invalid hvac_setpoint: {} (must be in [15, 30])", hvac_setpoint));
    }

    // Validate window U-value
    if window_u_value < 0.1 || window_u_value > 5.0 {
        return Err(format!("Invalid window_u_value: {} (must be in [0.1, 5.0])", window_u_value));
    }

    // Create ThermalModel if all validations pass
    Ok(ThermalModel {
        num_zones,
        temperatures: VectorField::zeros(num_zones * 8760),
        mass_temperatures: VectorField::zeros(num_zones * 8760),
        loads: VectorField::zeros(num_zones * 8760),
        window_u_value,
        hvac_setpoint,
        h_tr_em: VectorField::new(vec![h_tr_em; num_zones * 8760]),
        h_tr_ms: VectorField::new(vec![h_tr_ms; num_zones * 8760]),
        h_tr_is: VectorField::new(vec![h_tr_is; num_zones * 8760]),
        h_tr_w: VectorField::new(vec![h_tr_w; num_zones * 8760]),
        h_ve: VectorField::new(vec![h_ve; num_zones * 8760]),
    })
}

pub fn new_with_assembly_validation(
    num_zones: usize,
    assembly: &BuildingAssembly,
) -> Result<Self, String> {
    // Validate assembly configuration
    let assembly_result = validate_assembly(&assembly.to_yaml(), "assembly");
    if assembly_result.validation != "passed" {
        let errors: Vec<String> = assembly_result.errors
            .into_iter()
            .map(|e| format!("{}.layers[{}]: {}", e.path, e.message))
            .collect();
        return Err(format!("Assembly validation failed: {}", errors.join("; ")));
    }

    // Create ThermalModel with assembly
    Ok(Self::new_with_assembly(num_zones, assembly.clone()))
}
