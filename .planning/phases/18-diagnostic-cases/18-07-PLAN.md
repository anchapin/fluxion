    let matches = App::new("fluxion")
        .version(env!("CARGO_PKG_VERSION"))
        .about("Fluxion: Building Energy Modeling Engine")
        .subcommand(SubCommand::with_name("validate")
            .about("Run ASHRAE 140 validation")
            .arg(Arg::with_name("all")
                .long("all")
                .help("Run complete validation suite"))
            .arg(Arg::with_name("case")
                .long("case")
                .takes_value(true)
                .help("Validate specific case (e.g., 600)")))
        .get_matches();
    if let Some(matches) = matches.subcommand_matches("validate") {
        if matches.is_present("all") {
            // Run complete validation
            validate_analytical_engine();
        } else if let Some(case_num) = matches.value_of("case") {
            // Run specific case
            validate_single_case(case_num);
        } else {
            // Default: auto-discover and run baseline cases
            validate_analytical_engine();
        }
    }
}
