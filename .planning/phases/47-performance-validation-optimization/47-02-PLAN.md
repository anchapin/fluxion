    impl SolverOptimization {
        pub fn calculate_improvement(before: &PerformanceMetrics, after: &PerformanceMetrics) -> Self {
            let improvement = ((before.timestep_duration - after.timestep_duration) 
                / before.timestep_duration) * 100.0;

                before: before.clone(),
                after: after.clone(),
                improvement_percent: improvement,
            }
        }
    }
