   ```rust
   use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
   use fluxion::{Model, WeatherData, BuildingConfig};

   fn create_test_model() -> Model {
       // ASHRAE 140 Case 600 baseline
       let config = BuildingConfig::ashrae_140_case_600();
       let weather = WeatherData::from_epw("tests/data/ashrae_140/denver.epw");
       Model::new(config, weather)
   }

   fn bench_solve_timestep(c: &mut Criterion) {
       let mut model = create_test_model();

       c.bench_function("solve_timestep_single_zone", |b| {
           b.iter(|| {
               model.solve_timestep(black_box(0));
           })
       });
   }

   fn bench_annual_simulation(c: &mut Criterion) {
       let mut group = c.benchmark_group("annual_simulation");
       group.sample_size(10); // Fewer samples for long benchmarks
       group.measurement_time(std::time::Duration::from_secs(60));

       group.bench_function("simulate_annual_single_zone_8760", |b| {
           b.iter(|| {
               let mut model = create_test_model();
               model.simulate_full_year();
           })
       });
   }

   criterion_group!(benches, bench_solve_timestep, bench_annual_simulation);
   ```

   **benches/population_throughput.rs:**
   ```rust
   use criterion::{criterion_group, criterion_main, Criterion, Throughput};
   use fluxion::{BatchOracle, Model};

   fn create_config_variants(n: usize) -> Vec<Model> {
       (0..n).map(|i| {
           let mut config = BuildingConfig::ashrae_140_case_600();
           // Vary insulation thickness
           config.wall_insulation_thickness = 0.1 + (i as f64 * 0.01);
           Model::new(config, WeatherData::from_epw("tests/data/ashrae_140/denver.epw"))
       }).collect()
   }

   fn bench_population_throughput(c: &mut Criterion) {
       let mut group = c.benchmark_group("population_throughput");
       group.sample_size(10);
       group.measurement_time(std::time::Duration::from_secs(120));

       for n in [10, 100, 1000, 10000].iter() {
           let configs = create_config_variants(*n);

           group.throughput(Throughput::Elements(*n as u64));
           group.bench_with_input(
               BenchmarkId::from_parameter(n),
               &configs,
               |b, configs| {
                   b.iter(|| {
                       let oracle = BatchOracle::new();
                       oracle.simulate_batch(configs);
                   })
               },
           );
       }
       group.finish();
   }

   criterion_group!(benches, bench_population_throughput);
   ```

   **benches/ffi_overhead.rs:**
   ```rust
   use criterion::{black_box, criterion_group, criterion_main, Criterion};
   use fluxion::{Model, WeatherData, BuildingConfig};
   use pyo3::prelude::*;

   fn bench_native_rust(c: &mut Criterion) {
       let mut model = create_test_model();

       c.bench_function("rust_native_solve_timestep", |b| {
           b.iter(|| {
               model.solve_timestep(black_box(0));
           })
       });
   }

   fn bench_pyo3_boundary(c: &mut Criterion) {
       // Measure overhead of calling through PyO3
       Python::with_gil(|py| {
           let fluxion = PyModule::import(py, "fluxion").unwrap();
           let model = fluxion.call_method0("Model").unwrap();

           c.bench_function("pyo3_boundary_solve_timestep", |b| {
               b.iter(|| {
                   model.call_method1("solve_timestep", (0,)).unwrap();
               })
           });
       });
   }

   criterion_group!(benches, bench_native_rust, bench_pyo3_boundary);
   ```

### Task 3: Implement Performance Regression Detection

**Type:** implementation
**Time:** 5 minutes

**Actions:**
1. Create baseline storage:
   ```
   benches/baselines/
   ├── tier1_baseline.json    # Low-end hardware baseline
   ├── tier2_baseline.json    # Mid-range hardware baseline
   ├── tier3_baseline.json    # High-end hardware baseline
   └── tier4_baseline.json    # Server hardware baseline
   ```

2. Create regression detection script:
   ```rust
   // benches/check_regression.rs
   use std::fs;
   use serde::{Deserialize, Serialize};

   #[derive(Serialize, Deserialize)]
   struct BenchmarkBaseline {
       name: String,
       mean_time_ns: f64,
       throughput_configs_per_sec: Option<f64>,
   }

   #[derive(Serialize, Deserialize)]
   struct BenchmarkResult {
       name: String,
       mean_time_ns: f64,
       throughput_configs_per_sec: Option<f64>,
   }

   fn check_regression(baseline: &BenchmarkBaseline, current: &BenchmarkResult, threshold: f64) -> bool {
       let time_increase = (current.mean_time_ns - baseline.mean_time_ns) / baseline.mean_time_ns;
       time_increase > threshold  // e.g., 0.10 for 10% threshold
   }

   fn main() {
       // Load baseline
       let baseline: BenchmarkBaseline = serde_json::from_str(
           &fs::read_to_string("benches/baselines/tier3_baseline.json").unwrap()
       ).unwrap();

       // Load current results (from criterion output)
       let current: BenchmarkResult = serde_json::from_str(
           &fs::read_to_string("target/criterion/latest/results.json").unwrap()
       ).unwrap();

       // Check regression with 10% threshold
       if check_regression(&baseline, &current, 0.10) {
           eprintln!("❌ PERFORMANCE REGRESSION DETECTED");
           eprintln!("   Benchmark: {}", current.name);
           eprintln!("   Baseline: {:.2} ms", baseline.mean_time_ns / 1_000_000.0);
           eprintln!("   Current:  {:.2} ms", current.mean_time_ns / 1_000_000.0);
           eprintln!("   Change:   +{:.2}%", ((current.mean_time_ns / baseline.mean_time_ns) - 1.0) * 100.0);
           std::process::exit(1);
       } else {
           println!("✅ No performance regression detected");
       }
   }
   ```

3. Add CI integration script:
   ```bash
   # scripts/check_performance.sh
   #!/bin/bash
   set -e

   echo "Running benchmarks..."
   cargo bench --no-run  # Compile benchmarks

   # Run benchmarks and capture results
   cargo bench -- --save-baseline main

   # Check for regressions
   cargo run --bin check_regression

   echo "Performance check passed"
   ```

4. Add GitHub Actions workflow:
   ```yaml
   # .github/workflows/performance.yml
   name: Performance Regression Check

   on: [pull_request]

   jobs:
   performance:
       runs-on: ubuntu-latest
       steps:
       - uses: actions/checkout@v4

       - name: Install Rust
         uses: dtolnay/rust-action@stable

       - name: Run benchmarks
         run: ./scripts/check_performance.sh
   ```

**Verification:**
- [ ] Baseline files created with realistic values
- [ ] Regression detection script compiles and runs

- [ ] GitHub Actions workflow validates syntax

   ```rust
   fn bench_ashrae_140_suite(c: &mut Criterion) {

       // Simulate one optimization iteration (100 configs)
       group.throughput(Throughput::Elements(100));
       group.bench_function("optimization_iteration_100_configs", |b| {
           b.iter(|| {
               let oracle = BatchOracle::new();
               let configs = create_optimization_population(100);
               oracle.simulate_batch(configs);
           })
       });
   }

   fn bench_parametric_sweep(c: &mut Criterion) {

       group.sample_size(10);

       group.throughput(Throughput::Elements(50));

           b.iter(|| {
               let oracle = BatchOracle::new();
   ```
