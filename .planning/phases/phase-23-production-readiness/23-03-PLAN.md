   use thiserror::Error;

   #[derive(Error, Debug)]
   pub enum ValidationError {
       #[error("Parameter '{0}' must be positive, got {1}")]
       MustBePositive(String, f64),

       #[error("Parameter '{0}' must be in range [{1}, {2}], got {3}")]
       OutOfRange(String, f64, f64, f64),

       #[error("Parameter '{0}' must be finite, got {1}")]
       NotFinite(String, f64),

       #[error("Array '{0}' must have length {1}, got {2}")]
       WrongArrayLength(String, usize, usize),

       #[error("Weather data missing required field: {0}")]
       MissingWeatherField(String),

       #[error("Building config invalid: {0}")]
       InvalidBuildingConfig(String),
   }

   pub trait Validatable {
       fn validate(&self) -> Result<(), ValidationError>;
   }

   impl Validatable for BuildingConfig {
       fn validate(&self) -> Result<(), ValidationError> {
           // Thermal capacity must be positive
           if self.thermal_capacity <= 0.0 {
               return Err(ValidationError::MustBePositive(
                   "thermal_capacity".to_string(),
                   self.thermal_capacity,
               ));
           }

           // Thermal capacity must be in realistic range (10-1000 kWh/K)
           if !(10.0..=1000.0).contains(&self.thermal_capacity) {
               return Err(ValidationError::OutOfRange(
                   "thermal_capacity".to_string(),
                   10.0,
                   1000.0,
                   self.thermal_capacity,
               ));
           }

           // Wall U-value must be positive and realistic (0.1-10 W/m²K)
           if !(0.1..=10.0).contains(&self.wall_u_value) {
               return Err(ValidationError::OutOfRange(
                   "wall_u_value".to_string(),
                   0.1,
                   10.0,
                   self.wall_u_value,
               ));
           }

           // Window area ratio must be 0-1
           if !(0.0..=1.0).contains(&self.window_area_ratio) {
               return Err(ValidationError::OutOfRange(
                   "window_area_ratio".to_string(),
                   0.0,
                   1.0,
                   self.window_area_ratio,
               ));
           }

           // All parameters must be finite
           if !self.thermal_capacity.is_finite() {
               return Err(ValidationError::NotFinite(
                   "thermal_capacity".to_string(),
                   self.thermal_capacity,
               ));
           }

           Ok(())
       }
   }

   impl Validatable for WeatherData {
       fn validate(&self) -> Result<(), ValidationError> {
           // Must have 8760 timesteps (or 8784 for leap year)
           if self.hourly_data.len() != 8760 && self.hourly_data.len() != 8784 {
               return Err(ValidationError::WrongArrayLength(
                   "hourly_data".to_string(),
                   8760,
                   self.hourly_data.len(),
               ));
           }

           // Check for required fields
           if self.dry_bulb_temp.is_empty() {
               return Err(ValidationError::MissingWeatherField(
                   "dry_bulb_temp".to_string(),
               ));
           }

           // Temperature must be in realistic range (-50 to 60°C)
           for (i, temp) in self.dry_bulb_temp.iter().enumerate() {
               if !(-50.0..=60.0).contains(temp) {
                   return Err(ValidationError::OutOfRange(
                       format!("dry_bulb_temp[{}]", i),
                       -50.0,
                       60.0,
                       *temp,
                   ));
               }
           }

           Ok(())
       }
   }
   ```

   ```python
   """Input validation utilities for Fluxion."""

   from typing import Any
   from fluxion import Model, BatchOracle


   def validate_model_inputs(config: dict, weather: dict) -> list[str]:
       """
       Validate model inputs before simulation.

       Parameters
       ----------
       config : dict
           Building configuration dictionary
       weather : dict
           Weather data dictionary

       Returns
       -------
       list[str]
           List of validation errors (empty if valid)

       Examples
       --------
       >>> errors = validate_model_inputs(config, weather)
       >>> if errors:
       ...     for error in errors:
       ...         print(f"Validation error: {error}")
       ... else:
       ...     print("Inputs are valid")
       """
       errors = []

       # Check required config fields
       required_config_fields = [
           'thermal_capacity', 'wall_u_value', 'window_area_ratio',
           'floor_area', 'volume'
       ]
       for field in required_config_fields:
           if field not in config:
               errors.append(f"Missing required config field: {field}")

       # Check value ranges
       if 'thermal_capacity' in config:
           tc = config['thermal_capacity']
           if tc <= 0:
               errors.append(f"thermal_capacity must be positive, got {tc}")
           elif not (10.0 <= tc <= 1000.0):
               errors.append(
                   f"thermal_capacity should be 10-1000 kWh/K, got {tc}"
               )

       # Check weather data
       if 'hourly_data' in weather:
           data = weather['hourly_data']
           if len(data) not in [8760, 8784]:
               errors.append(
                   f"Weather data must have 8760 or 8784 timesteps, "
                   f"got {len(data)}"
               )

       return errors
   ```

### Task 2: Document Failure Modes

**Type:** documentation
**Time:** 5 minutes

**Actions:**
1. Create failure modes documentation:

   **docs/user-guide/failure_modes.md:**
   ```markdown
   # Failure Modes Documentation

   This document describes what happens when things go wrong in Fluxion.

   ## Input Validation Failures

   ### Invalid Building Configuration

   **Scenario:** User provides invalid building parameters

   **Example:**
   ```python
   >>> from fluxion import Model
   >>> config = {'thermal_capacity': -5.0}  # Invalid: negative
   >>> model = Model(config, weather)
   ValueError: Parameter 'thermal_capacity' must be positive, got -5.0
   ```

   **Behavior:**
   - Constructor raises `ValueError` immediately
   - Model is NOT created
   - Error message specifies the invalid parameter and expected range

   ### Invalid Weather Data

   **Scenario:** Weather data has wrong number of timesteps

   **Example:**
   ```python
   >>> weather = {'hourly_data': [20.0] * 5000}  # Only 5000 timesteps
   >>> model = Model(config, weather)
   ValueError: Array 'hourly_data' must have length 8760, got 5000
   ```

   **Behavior:**
   - Constructor raises `ValueError` immediately
   - Suggests checking EPW file or weather data source

   ## Simulation Failures

   ### Numerical Instability

   **Scenario:** Thermal network solver fails to converge

   **Example:**
   ```python
   >>> model = Model(config, weather)
   >>> model.solve_timestep(4320)  # Mid-summer peak load
   RuntimeError: Solver failed to converge after 100 iterations
   ```

   **Behavior:**
   - Raises `RuntimeError` with convergence details
   - Suggests checking timestep parameters or building configuration
   - Model state remains unchanged (transactional)

   ### HVAC System Failure

   **Scenario:** HVAC capacity insufficient for load

   **Example:**
   ```python
   >>> results = model.simulate_full_year()
   >>> # Warning logged: "HVAC capacity exceeded at timestep 4320"
   ```

   **Behavior:**
   - Logs warning (does not fail)
   - Continues simulation with degraded performance
   - Records unmet load hours in results

   ## FFI Boundary Failures

   ### NumPy Array Conversion Error

   **Scenario:** Python passes invalid array to Rust

   **Example:**
   ```python
   >>> import numpy as np
   >>> model.set_weather_data(np.array([20.0, np.nan, 25.0]))
   ValueError: Array contains NaN values at indices: [1]
   ```

   **Behavior:**
   - Validates array before conversion
   - Raises `ValueError` with specific indices
   - Rust code never sees invalid data

   ### Memory Allocation Failure

   **Scenario:** System runs out of memory during large batch

   **Example:**
   ```python
   >>> oracle.simulate_batch(1000000_configs)  # Too many configs
   MemoryError: Failed to allocate 50GB for batch simulation
   ```

   **Behavior:**
   - Raises `MemoryError` (Python exception)
   - Cleans up partial allocations
   - Suggests reducing batch size

   ## Error Recovery Strategies

   ### Graceful Degradation

   When HVAC capacity is exceeded:
   1. Log warning with timestep and load details
   2. Continue simulation with available capacity
   3. Track unmet load hours in results
   4. Report in summary: "Warning: 23 hours with unmet cooling load"

   ### Transactional Simulation

   Each timestep is atomic:
   1. Save state before timestep
   2. Attempt solve
   3. On success: commit state
   4. On failure: restore previous state, raise error

   ### Batch Processing with Failures

   For batch simulations:
   1. Process configs in independent batches
   2. If config #47 fails, configs #1-46 and #48-100 still complete
   3. Return results with failure metadata:
      ```python
      results = {
          'successful': 99,
          'failed': 1,
          'failures': [{'config_id': 47, 'error': 'Solver divergence'}]
      }
      ```

   ## Panic Prevention

   Fluxion guarantees **no panics in release builds**:

   - All `unwrap()` calls replaced with proper error handling
   - All `expect()` calls have actionable messages
   - All `panic!()` calls replaced with `Result` returns
   - `#[cfg(not(debug_assertions))]` ensures release mode safety

   **Exception:** Debug builds may panic on assertion failures to catch bugs early.
   ```

       pub successful: usize,
       pub failed: usize,
       pub results: Vec<SimulationResult>,
       pub failures: Vec<FailureInfo>,
   }

   pub struct FailureInfo {
       pub config_id: usize,
       pub error: String,
   }

   impl BatchOracle {
       pub fn simulate_batch(&self, configs: Vec<Model>) -> BatchResult {
           let mut results = Vec::new();
           let mut failures = Vec::new();

           for (i, config) in configs.into_iter().enumerate() {
               match self.simulate_single(config) {
                   Ok(result) => results.push(result),
                   Err(e) => failures.push(FailureInfo {
                       config_id: i,
                       error: e.to_string(),
                   }),
               }
           }

           BatchResult {
               successful: results.len(),
               failed: failures.len(),
               results,
               failures,
           }
       }
   }

### Task 4: Document Determinism Guarantees

**Type:** documentation
**Time:** 5 minutes

**Actions:**
1. Create determinism documentation:

   **docs/user-guide/determinism.md:**
   ```markdown
   # Determinism Guarantees

   Fluxion guarantees **deterministic results**: same inputs always produce same outputs.

   ## What This Means

   Given:
   - Identical building configuration
   - Identical weather data
   - Identical simulation parameters
   - Same Fluxion version

   You will always get:
   - Identical hourly energy consumption (to floating-point precision)
   - Identical HVAC loads
   - Identical zone temperatures

   ## Why This Matters

   ### Optimization Reproducibility

   Optimization algorithms (genetic algorithms, gradient descent) require deterministic evaluations:

   ```python
   >>> from fluxion import BatchOracle
   >>> oracle = BatchOracle()
   >>>
   >>> # First evaluation
   >>> result1 = oracle.evaluate(config)
   >>> # Second evaluation (same config)
   >>> result2 = oracle.evaluate(config)
   >>>
   >>> assert result1.energy == result2.energy  # Always true
   ```

   Without determinism, optimization would be unreliable.

   ### Debugging and Validation

   Determinism enables:
   - Reproducing bugs (same inputs → same failure)
   - Validating fixes (same inputs → improved outputs)
   - Regression testing (compare against baseline)

   ## How We Guarantee Determinism

   ### No Random Number Generation

   Fluxion does NOT use:
   - `rand::random()` without explicit seed
   - Hash map iteration order (uses BTreeMap for ordered iteration)
   - Parallel reduction without deterministic ordering

   ### Consistent Floating-Point Operations

   We ensure deterministic floating-point by:
   - Using `f64` consistently (not mixing f32/f64)
   - Avoiding parallel reduction for summations (or using ordered parallel sum)
   - Documenting platform-specific behavior (x86 vs ARM may differ in last bits)

   ### Thread Safety Without Race Conditions

   All mutable state is:
   - Protected by locks (Mutex, RwLock)
   - Or confined to single thread during simulation

   ## Known Limitations

   ### Cross-Platform Variations

   **Issue:** x86 and ARM may produce slightly different results (last 2-3 bits)

   **Reason:** Different floating-point unit implementations

   **Impact:** Results match to ~15 significant digits (f64 precision)

   **Mitigation:** Use tolerance-based comparison in tests:
   ```rust
   assert_relative_eq!(result1, result2, epsilon = 1e-10);
   ```

   ### Parallel Execution Order

   **Issue:** Batch simulations may process configs in different order

   **Reason:** Parallel iterator scheduling

   **Impact:** Individual config results identical, but batch result array order may vary

   **Mitigation:** Sort results by config_id if order matters:
   ```python
   results.sort(key=lambda r: r.config_id)
   ```

   ## Testing Determinism

   We test determinism with:

   ```rust
   #[test]
   fn test_determinism_single_config() {
       let config = BuildingConfig::ashrae_140_case_600();
       let weather = WeatherData::from_epw("denver.epw");

       // Run simulation twice
       let mut model1 = Model::new(config.clone(), weather.clone());
       model1.simulate_full_year();

       let mut model2 = Model::new(config.clone(), weather.clone());
       model2.simulate_full_year();

       // Results must be identical
       assert_eq!(model1.annual_energy(), model2.annual_energy());
   }

   #[test]
   fn test_determinism_batch() {
       let oracle = BatchOracle::new();
       let configs = create_test_population(100);

       // Run batch twice
       let results1 = oracle.simulate_batch(configs.clone());
       let results2 = oracle.simulate_batch(configs.clone());

       // All results must match
       for (r1, r2) in results1.iter().zip(results2.iter()) {
           assert_eq!(r1.energy, r2.energy);
       }
   }
   ```

   ## Version-to-Version Determinism

   **Policy:** Minor versions (v0.5.x) maintain determinism with previous minor versions.

   Major versions (v0.x → v1.0) may change physics algorithms, breaking determinism.

   **Migration:** When determinism breaks, we provide:
   - Migration guide with expected differences
   - Side-by-side comparison tool
   - Deprecation period (one minor version)
   ```
