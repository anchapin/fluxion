fn test_ashrae_140_statistical_compliance() {

    let cases = vec!["600", "610", "620", "630", "640", "650",  // Baseline
                     "900", "910", "920", "930", "940", "950", "960",  // HighMass
                     "600FF", "650FF", "900FF", "950FF",  // FreeFloating
                     "800", "801", "802", "803", "804", "805", "806", "807", "808", "809", "810"];  // Equipment

    let report = validator.validate_with_statistics(&cases).unwrap();

    // Verify group validation results
    let baseline_pass = report.group_validation.get(&ValidationGroup::Baseline);
    assert!(baseline_pass.is_some());

    // Verify statistical metrics are finite
    assert!(report.metrics.nmbe.is_finite());
    assert!(report.metrics.cv_rmse.is_finite());

    // Generate report to verify formatting
    let generator = ValidationReportGenerator::new();
    let report_md = generator.generate_report_with_statistics(&report.tolerance);
    assert!(report_md.contains("## Statistical Validation"));
}
