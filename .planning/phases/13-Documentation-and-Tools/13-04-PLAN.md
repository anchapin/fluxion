   use std::f64::consts::SQRT_2;

   pub struct DeltaTestResult {
       pub metric_name: String,
       pub delta_value: f64,
       pub p_value: Option<f64>,  // Statistical significance
       pub is_significant: bool,
       pub confidence_interval: Option<(f64, f64)>,
   }

   pub fn perform_delta_test(
       baseline: &[f64],
       current: &[f64],
       confidence_level: f64,  // 0.95 for 95% confidence
   ) -> DeltaTestResult {
       let n = baseline.len();
       let m = current.len();

       // Calculate means
       let mean_baseline: f64 = baseline.iter().sum::<f64>() / n as f64;
       let mean_current: f64 = current.iter().sum::<f64>() / m as f64;

       // Calculate standard deviations
       let std_baseline = std_deviation(baseline, mean_baseline);
       let std_current = std_deviation(current, mean_current);

       // Calculate delta (mean difference)
       let delta = mean_current - mean_baseline;

       // Calculate pooled standard error
       let pooled_std_error = ((std_baseline.powi(2) / n as f64) + (std_current.powi(2) / m as f64)).sqrt();

       // Calculate z-score (for large samples)
       let z_score = if pooled_std_error > 0.0 {
           delta / pooled_std_error
       } else {
           0.0
       };

       // Calculate p-value (two-tailed test)
       let p_value = if z_score.abs() > 0.0 {
           2.0 * (1.0 - normal_cdf(z_score.abs()))
       } else {
           1.0
       };

       // Determine significance (p < 0.05 for 95% confidence)
       let is_significant = p_value < 0.05;

       // Calculate confidence interval
       let critical_value = 1.96;  // For 95% confidence
       let ci_lower = delta - critical_value * pooled_std_error;
       let ci_upper = delta + critical_value * pooled_std_error;

       DeltaTestResult {
           metric_name: "delta".to_string(),
           delta_value: delta,
           p_value: Some(p_value),
           is_significant,
           confidence_interval: Some((ci_lower, ci_upper)),
       }
   }

   fn std_deviation(data: &[f64], mean: f64) -> f64 {
       let variance: f64 = data.iter()
           .map(|&x| (x - mean).powi(2))
           .sum::<f64>() / (data.len() - 1) as f64;
       variance.sqrt()
   }

   fn normal_cdf(x: f64) -> f64 {
       // Approximation of standard normal CDF
       let sign = if x < 0.0 { -1.0 } else { 1.0 };
       let a1 = 0.254829592;
       let a2 = -0.284496736;
       let a3 = 1.421413741;
       let a4 = -1.453152027;
       let a5 = 1.061405429;
       let p = 0.3275911;

       let k = 1.0 / (1.0 + p * x.abs());
       let y = 1.0 - (((((a5 * k + a4) * k) + a3) * k + a2) * k + a1) * k * (-0.5 * x * x).exp();

       0.5 * (1.0 + sign * y)
   }
   ```

   pub struct SensitivityResult {
       pub parameter_name: String,
       pub coefficient: f64,           // Raw coefficient
       pub normalized_coefficient: f64,  // Normalized by parameter range
       pub ranking: usize,              // Parameter ranking (1 = most sensitive)
   }

   pub fn normalize_sensitivity(
       sensitivity_results: &[(String, f64)],
       parameter_ranges: &HashMap<String, (f64, f64)>,  // (min, max)
   ) -> Vec<SensitivityResult> {
       let mut normalized_results = Vec::new();

       for (param_name, coefficient) in sensitivity_results {
           // Get parameter range
           let (min_val, max_val) = parameter_ranges.get(param_name)
               .unwrap_or(&(0.0, 1.0));

           let range = max_val - min_val;

           // Normalize coefficient by dividing by parameter range
           let normalized_coefficient = if range > 0.0 {
               coefficient.abs() / range
           } else {
               coefficient.abs()
           };

           normalized_results.push(SensitivityResult {
               parameter_name: param_name.clone(),
               coefficient: *coefficient,
               normalized_coefficient,
               ranking: 0,  // Will be assigned after sorting
           });
       }

       // Sort by normalized coefficient (descending)
       normalized_results.sort_by(|a, b| {
           b.normalized_coefficient.partial_cmp(&a.normalized_coefficient).unwrap()
       });

       // Assign rankings
       for (i, result) in normalized_results.iter_mut().enumerate() {
           result.ranking = i + 1;
       }

       normalized_results
   }
   ```

   ) -> Vec<SensitivityResult> {

   fn calculate_raw_sensitivity(
       model: &ThermalModel,
       base_params: &[f64],
   ) -> Vec<(String, f64)> {
       let mut results = Vec::new();
       let base_energy = model.solve_timesteps(8760, &surrogates, false);

       // Test each parameter with small perturbation
       for (i, &param_value) in base_params.iter().enumerate() {
           let mut perturbed_params = base_params.to_vec();
           perturbed_params[i] = param_value * 1.01;  // 1% perturbation

           model.apply_parameters(&perturbed_params);
           let perturbed_energy = model.solve_timesteps(8760, &surrogates, false);

           // Calculate sensitivity coefficient
           let delta_energy = perturbed_energy - base_energy;
           let coefficient = delta_energy / (param_value * 0.01);  // dE/dp

           results.push((format!("param_{}", i), coefficient));
       }

       results
   }
