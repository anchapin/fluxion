    ```rust
    use clap::{Arg, ArgMatches, Subcommand};
    use crate::validation::ashrae140::{ASHRAE140Case, run_validation};

    #[derive(Subcommand, Debug)]
    pub enum ValidationSubcommand {
        /// Run a single ASHRAE 140 case
        #[clap(name = "run")]
        Run {
            /// ASHRAE 140 case to run (e.g., 800, 900, 195)
            case: u32,
            /// Output directory for results
            #[clap(short, long, default_value = "./results")]
            output: String,
            /// Run in verbose mode
            #[clap(short, long)]
            verbose: bool,
        },

        /// Run a series of ASHRAE 140 cases
        #[clap(name = "run-series")]
        RunSeries {
            /// Series to run (800-810 or 195-470)
            series: String,
            /// Output directory for results
            #[clap(short, long, default_value = "./results")]
            output: String,
            /// Run in verbose mode
            #[clap(short, long)]
            verbose: bool,
        },

        /// List available ASHRAE 140 cases
        #[clap(name = "list-cases")]
        ListCases,
    }

    pub fn handle_validation_command(command: &ValidationSubcommand) -> Result<()> {
        match command {
            ValidationSubcommand::Run { case, output, verbose } => {
                let case = parse_case_number(*case)?;
                run_single_case(case, output, *verbose)
            },
            ValidationSubcommand::RunSeries { series, output, verbose } => {
                let cases = parse_series(series)?;
                run_case_series(&cases, output, *verbose)
            },
            ValidationSubcommand::ListCases => {
                list_available_cases()
            },
        }
    }

    fn parse_case_number(case_num: u32) -> Result<ASHRAE140Case> {
        match case_num {
            800 => Ok(ASHRAE140Case::Case800),
            801 => Ok(ASHRAE140Case::Case801),
            // ... all cases through 810 and 195-470
            470 => Ok(ASHRAE140Case::Case470),
            _ => Err(anyhow!("Case {} not supported. Use --list-cases to see available cases.", case_num)),
        }
    }

    fn parse_series(series: &str) -> Result<Vec<ASHRAE140Case>> {
        match series {
            "800-810" | "hvac" => Ok((800..=810).map(|n| {
                match n {
                    800 => ASHRAE140Case::Case800,
                    // ... map all cases
                    810 => ASHRAE140Case::Case810,
                    _ => unreachable!(),
                }
            }).collect()),
            "195-470" | "diagnostic" => Ok((195..=470).map(|n| {
                match n {
                    195 => ASHRAE140Case::Case195,
                    // ... map all cases
                    470 => ASHRAE140Case::Case470,
                    _ => unreachable!(),
                }
            }).collect()),
            _ => Err(anyhow!("Invalid series. Use '800-810' or '195-470'")),
        }
    }
    ```

    ```rust
    fn run_cross_validation(
        case_num: u32,
        tool: String,
        reference_file: String,
        output_dir: String,
        tolerance: Option<f64>,
        detailed: bool,
    ) -> Result<()> {
        // Parse case number
        let case = parse_case_number(case_num)?;

        // Create appropriate adapter
        let adapter: Box<dyn CrossValidationAdapter> = match tool.as_str() {
            "energyplus" => Box::new(EnergyPlusAdapter),
            "trnsys" => Box::new(TRNSYSAdapter),
            _ => return Err(anyhow!("Invalid tool. Use 'energyplus' or 'trnsys'")),
        };

        // Load Fluxion results (run case if not cached)
        let fluxion_results = run_validation(case)?;

        // Load reference results
        let reference_results = adapter.load_reference_results(case, &reference_file)?;

        // Compare results
        let comparison = if let Some(tol) = tolerance {
            adapter.compare_results_with_tolerance(&fluxion_results, &reference_results, tol)
        } else {
            adapter.compare_results(&fluxion_results, &reference_results)
        };

        // Generate report
        let report = adapter.generate_report(&comparison);

        // Save report to file
        save_report(&report, &output_dir, case, adapter.tool_name())?;

        // Print summary
        println!("{}", report);

        Ok(())
    }
    ```

    ```rust
    fn run_batch_cross_validation(
        series: &str,
        tool: &str,
        reference_dir: &str,
        output_dir: &str,
        parallel: usize,
    ) -> Result<()> {
        let cases = parse_series(series)?;
        let adapter: Box<dyn CrossValidationAdapter> = match tool {
            "energyplus" => Box::new(EnergyPlusAdapter),
            "trnsys" => Box::new(TRNSYSAdapter),
            _ => return Err(anyhow!("Invalid tool")),
        };

        // Use Rayon for parallel processing
        cases.par_iter()
            .for_each(|case| {
                let reference_file = format!("{}/case_{:03}.csv", reference_dir, case as u32);
                if let Ok(reference_path) = std::path::PathBuf::from(&reference_file).canonicalize() {
                    if let Err(e) = run_cross_validation(
                        case as u32,
                        tool.to_string(),
                        reference_path.to_string_lossy().into_owned(),
                        output_dir.to_string(),
                        None,
                        false
                    ) {
                        eprintln!("Failed to validate case {:?}: {}", case, e);
                    }
                }
            });

        Ok(())
    }
    ```
