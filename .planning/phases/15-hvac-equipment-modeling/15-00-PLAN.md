    /// HVAC operating mode for efficiency curve selection
    #[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
    pub enum HVACMode {
        Heating,
        Cooling,
        Off,
    }

    impl From<HeatPumpMode> for HVACMode {
        fn from(mode: HeatPumpMode) -> Self {
            match mode {
                HeatPumpMode::Heating => HVACMode::Heating,
                HeatPumpMode::Cooling => HVACMode::Cooling,
                HeatPumpMode::Off => HVACMode::Off,
            }
        }
    }

    /// Trait for variable-capacity HVAC equipment with efficiency curves.
    ///
    /// This trait defines a unified interface for all equipment types that support
    /// continuous modulation from 0-100% capacity, including VAV, CAV, heat pumps,
    /// chillers, and boilers.
    ///
    /// Key concepts:
    /// - Part-Load Ratio (PLR): Actual load / rated capacity, range [0.0, 1.0]
    /// - Variable capacity: Equipment can modulate output continuously (not just on/off)
    /// - Efficiency curves: COP/efficiency varies with PLR and outdoor temperature
    ///
    /// TODO: Implement full trait in Plan 15-01
    pub trait VariableCapacityEquipment: Send + Sync + Clone {
        /// Calculate actual capacity at current conditions (W)
        ///
        /// # Arguments
        /// * `plr` - Part-load ratio (0.0 to 1.0)
        /// * `outdoor_temp` - Outdoor air temperature (°C)
        ///
        /// # Returns
        /// Actual capacity in Watts, accounting for temperature-dependent capacity degradation
        fn calculate_capacity(&self, plr: f64, outdoor_temp: f64) -> f64;

        /// Calculate efficiency (COP or EER) at current conditions
        ///
        /// # Arguments
        /// * `plr` - Part-load ratio (0.0 to 1.0)
        /// * `outdoor_temp` - Outdoor air temperature (°C)
        /// * `mode` - HVAC operating mode (heating or cooling)
        ///
        /// # Returns
        /// Coefficient of Performance (COP) or Energy Efficiency Ratio (EER)
        fn calculate_efficiency(&self, plr: f64, outdoor_temp: f64, mode: HVACMode) -> f64;

        /// Calculate power consumption at current conditions (W)
        ///
        /// # Arguments
        /// * `load` - Required thermal load (W)
        /// * `outdoor_temp` - Outdoor air temperature (°C)
        /// * `mode` - HVAC operating mode (heating or cooling)
        ///
        /// # Returns
        /// Electrical power consumption in Watts
        fn calculate_power(&self, load: f64, outdoor_temp: f64, mode: HVACMode) -> f64;

        /// Get rated capacity at design conditions (W)
        ///
        /// Returns maximum heating or cooling capacity (whichever is larger)
        fn rated_capacity(&self) -> f64;

        /// Get rated efficiency at design conditions
        ///
        /// # Arguments
        /// * `mode` - HVAC operating mode (heating or cooling)
        ///
        /// # Returns
        /// Rated COP (heating) or EER (cooling) at design conditions
        fn rated_efficiency(&self, mode: HVACMode) -> f64;

        /// Get current part-load ratio (0.0 to 1.0)
        ///
        /// Returns the PLR from the most recent state update
        fn current_plr(&self) -> f64;

        /// Update equipment state for next timestep
        ///
        /// # Arguments
        /// * `current_load` - Thermal load applied this timestep (W)
        /// * `outdoor_temp` - Outdoor air temperature (°C)
        /// * `mode` - HVAC operating mode (heating or cooling)
        fn update_state(&mut self, current_load: f64, outdoor_temp: f64, mode: HVACMode);
    }

    /// Placeholder Chiller structure (full implementation in Plan 15-02)
