    #[pyo3(get, set)]

    pub threshold_hours: f64,

    #[pyo3(get, set)]
    pub h_interior: f64,

    #[pyo3(get, set)]
    pub h_exterior: f64,

    #[pyo3(get, set)]
    pub ctf_history_size: usize,

    #[pyo3(get, set)]
    pub fd_nodes_per_layer: usize,
}

#[pymethods]
impl PySolverConfig {
    #[new]
    fn new() -> Self {
        Self::default()
    }

    /// Validate configuration
    fn validate(&self) -> PyResult<()> {
        if self.threshold_hours < 0.5 || self.threshold_hours > 10.0 {
            return Err(PyValueError::new_err(
                "threshold_hours must be between 0.5 and 10.0"
            ));
        }
        if self.fd_nodes_per_layer < self.fd_min_nodes ||
           self.fd_nodes_per_layer > self.fd_max_nodes {
            return Err(PyValueError::new_err(
                "fd_nodes_per_layer must be between min and max"
            ));
        }
        Ok(())
    }
}

impl Default for PySolverConfig {
    fn default() -> Self {
        Self {
            method: "auto".to_string(),
            threshold_hours: 2.0,
            enable_fallback: true,
            h_interior: 8.0,
            h_exterior: 25.0,
            ctf_history_size: 50,
            fd_nodes_per_layer: 10,
        }
    }
}

### Task 4: Add Configuration Validation

**Create:** `src/config/solver_config.rs`

```rust
/// Solver configuration with validation
#[derive(Debug, Clone)]
pub struct SolverConfig {
    pub method: SolverMethod,
    pub threshold_hours: f64,
    pub enable_fallback: bool,
    pub h_interior: f64,
    pub h_exterior: f64,
    pub ctf_history_size: usize,
    pub fd_nodes_per_layer: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SolverMethod {
    Auto,
    FiveR1C,
    CTF,
    FiniteDifference,
}

impl SolverConfig {
    /// Validate configuration and return errors
    pub fn validate(&self) -> Result<(), ConfigError> {
        if self.threshold_hours < 0.5 || self.threshold_hours > 10.0 {
            return Err(ConfigError::InvalidThreshold(self.threshold_hours));
        }
        if self.h_interior < 1.0 || self.h_interior > 50.0 {
            return Err(ConfigError::InvalidHInterior(self.h_interior));
        }
        if self.h_exterior < 5.0 || self.h_exterior > 100.0 {
            return Err(ConfigError::InvalidHExterior(self.h_exterior));
        }
        if self.ctf_history_size < 10 || self.ctf_history_size > 200 {
            return Err(ConfigError::InvalidCTFHistory(self.ctf_history_size));
        }
        if self.fd_nodes_per_layer < 3 || self.fd_nodes_per_layer > 100 {
            return Err(ConfigError::InvalidFDDiscretization(self.fd_nodes_per_layer));
        }
        Ok(())
    }

    /// Parse method from string
    pub fn parse_method(s: &str) -> Result<SolverMethod, ConfigError> {
        match s.to_lowercase().as_str() {
            "auto" => Ok(SolverMethod::Auto),
            "5r1c" | "five_r1c" => Ok(SolverMethod::FiveR1C),
            "ctf" => Ok(SolverMethod::CTF),
            "fd" | "finite_difference" => Ok(SolverMethod::FiniteDifference),
            _ => Err(ConfigError::UnknownMethod(s.to_string())),
        }
    }
}
```

**Verification:**
- Validation catches invalid configurations
- Error messages are clear and actionable

### Task 6: Add Usage Examples

```python
#!/usr/bin/env python3
"""
Solver Configuration Examples

Demonstrates how to configure CTF/FD solvers in Fluxion.
"""

import fluxion

# Example 1: Automatic solver selection (default)
config = fluxion.ModelConfig()
config.solver_method = "auto"  # Auto-select based on thermal mass
config.solver_threshold_hours = 2.0  # τ > 2h → CTF/FD

model = fluxion.Model(config)
results = model.simulate(weather_file="USA_CA_San.Francisco.epw")

# Example 2: Force CTF solver (for high-mass buildings)
config = fluxion.ModelConfig()
config.solver_method = "ctf"
config.ctf_history_size = 50  # More history = more accurate, more memory

model = fluxion.Model(config)
results = model.simulate(weather_file="USA_CA_San.Francisco.epw")

# Example 3: Force FD solver (for complex constructions)
config = fluxion.ModelConfig()
config.solver_method = "fd"
config.fd_nodes_per_layer = 20  # Finer discretization = more accurate

model = fluxion.Model(config)
results = model.simulate(weather_file="USA_CA_San.Francisco.epw")

# Example 4: Custom convective coefficients
config = fluxion.ModelConfig()
config.solver_h_interior = 8.0  # Typical interior
config.solver_h_exterior = 25.0  # Typical exterior (windy)

model = fluxion.Model(config)
results = model.simulate(weather_file="USA_CA_San.Francisco.epw")

# Example 5: Disable fallback (debugging)
config = fluxion.ModelConfig()
config.solver_method = "ctf"
config.solver_fallback = False  # Don't fall back to FD on CTF failure

try:
    model = fluxion.Model(config)
    results = model.simulate(weather_file="USA_CA_San.Francisco.epw")
except Exception as e:
    print(f"CTF failed: {e}")
```

/// Solver method: "auto", "5r1c", "ctf", or "fd"
///
/// - "auto" (default): Automatically select based on thermal mass
///   - τ < 2 hours: 5R1C (fast, accurate for low-mass)
///   - τ ≥ 2 hours: CTF (accurate for high-mass)
///   - CTF fails: FD (fallback for extreme constructions)
///
/// - "5r1c": Force 5R1C thermal network (fastest, low-mass only)
///
/// - "ctf": Force Conduction Transfer Functions (accurate, high-mass)
///
/// - "fd": Force Finite Difference (most robust, slowest)
#[arg(long, default_value = "auto", help_heading = "Solver Configuration")]
pub solver_method: String,

/// Thermal mass threshold for automatic solver selection (hours)
///
/// Walls with time constant τ > threshold use CTF/FD instead of 5R1C.
///
/// Typical values:
/// - 1.5 hours: More aggressive CTF usage (higher accuracy)
/// - 2.0 hours: Default (balanced)
/// - 3.0 hours: Conservative CTF usage (faster, may lose accuracy)
#[arg(long, default_value = "2.0", help_heading = "Solver Configuration")]
pub solver_threshold_hours: f64,
