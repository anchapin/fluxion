/// Test CTF heat flux calculation
#[test]
fn test_ctf_flux_calculation() {
    let layers = vec![
        CTFMaterial::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];

    let coeffs = CTFCalculator::with_defaults(&layers, 3600.0).compute_coefficients();
    let config = CTFSolverConfig::new(3600.0, 50);
    let mut solver = CTFSolver::new(coeffs, config);

    // Apply constant temperatures
    let T_int = 20.0;
    let T_ext = 0.0;

    // Run for 24 hours to establish history
    let mut flux_history = Vec::new();
    for _ in 0..24 {
        let q_flux = solver.step(T_int, T_ext);
        flux_history.push(q_flux);
    }

    // Verify flux converges to steady-state
    let steady_state = (T_ext - T_int) / (1.0/8.0 + 0.2/1.4 + 1.0/25.0);
    let final_flux = flux_history[23];

    assert!(
        (final_flux - steady_state).abs() < 5.0,
        "Flux should converge to steady-state within 5 W/m²"
    );
}

fn test_ctf_step_response() {

        CTFMaterial::new("Concrete", 0.200, 1.4, 2300.0, 880.0),

/// Test FD steady-state temperature profile (should be linear)
#[test]
fn test_fd_steady_state_linear_profile() {
    let layers = vec![
        MaterialLayer::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];

    let disc = WallDiscretization::from_layers(&layers, 10);
    let mut solver = ImplicitFDSolver::new(disc, 20.0);

    let interior_bc = SurfaceBC::new_interior(8.0, 20.0);
    let exterior_bc = SurfaceBC::new_exterior(25.0, 0.0);

    // Run until steady-state (100 hours)
    for _ in 0..100 {
        solver.step(3600.0, &interior_bc, &exterior_bc);
    }

    // Get temperature profile
    let temps = solver.temperatures();

    // Verify linear profile (T(x) = T_int + (T_ext - T_int) * x / L)
    let L = 0.200;
    for i in 0..temps.len() {
        let x = disc.node_position(i);
        let expected = 20.0 + (0.0 - 20.0) * x / L;
        assert!(
            (temps[i] - expected).abs() < 0.5,
            "Temperature profile should be linear at steady-state"
        );
    }
}

#[test]
fn test_fd_steady_state_flux() {
    let layers = vec![
        MaterialLayer::new("Concrete", 0.200, 1.4, 2300.0, 880.0),

    let interior_bc = SurfaceBC::new_interior(8.0, 20.0);

    // Run until steady-state
    for _ in 0..100 {
        solver.step(3600.0, &interior_bc, &exterior_bc);

    );
}

### Task 6: Energy Conservation Tests

```rust
/// Test FD energy conservation over 24 hours
#[test]
fn test_fd_energy_conservation() {
    let layers = vec![
        MaterialLayer::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];

    let disc = WallDiscretization::from_layers(&layers, 10);
    let mut solver = ImplicitFDSolver::new(disc, 20.0);

    let interior_bc = SurfaceBC::new_interior(8.0, 20.0);
    let exterior_bc = SurfaceBC::new_exterior(25.0, 10.0);  // Warm exterior

    let mut energy_in = 0.0;  // J/m²
    let mut energy_out = 0.0;  // J/m²

    for _ in 0..24 {
        let q_in = solver.surface_flux();  // W/m² (positive = into wall)
        energy_in += q_in * 3600.0;  // Convert to J/m²

        solver.step(3600.0, &interior_bc, &exterior_bc);

        let q_out = solver.surface_flux_exterior();  // W/m² (positive = out of wall)
        energy_out += q_out * 3600.0;
    }

    // Calculate change in internal energy
    let temps_initial = solver.temperatures();

    // Run one more step to get final state
    solver.step(3600.0, &interior_bc, &exterior_bc);
    let temps_final = solver.temperatures();

    let mut delta_u = 0.0;  // J/m²
    for i in 0..temps_initial.len() {
        let dx = disc.node_spacing(i);
        let rho = disc.node_density(i);
        let cp = disc.node_specific_heat(i);
        let dT = temps_final[i] - temps_initial[i];

        delta_u += rho * cp * dx * dT;  // J/m²
    }

    // Energy balance: energy_in - energy_out = delta_u
    let imbalance = (energy_in - energy_out - delta_u).abs();
    assert!(
        imbalance < 100.0,  // Allow 100 J/m² tolerance
        "Energy balance should close within 100 J/m² (got {} J/m²)",
        imbalance
    );
}
```

**Verification:**
- Energy balance closes within tolerance
- No energy is created or destroyed

/// Compare CTF vs FD for simple wall
#[test]
fn test_ctf_vs_fd_simple_wall() {
    let layers = vec![
        CTFMaterial::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];

    // CTF solver
    let coeffs = CTFCalculator::with_defaults(&layers, 3600.0).compute_coefficients();
    let mut ctf_solver = CTFSolver::new(coeffs, CTFSolverConfig::new(3600.0, 50));

    // FD solver
    let fd_layers = vec![
        MaterialLayer::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];
    let disc = WallDiscretization::from_layers(&fd_layers, 20);
    let mut fd_solver = ImplicitFDSolver::new(disc, 20.0);

    let T_int = 20.0;
    let T_ext = 0.0;

    // Run both solvers for 24 hours
    let mut ctf_fluxes = Vec::new();
    let mut fd_fluxes = Vec::new();

    for _ in 0..24 {
        ctf_fluxes.push(ctf_solver.step(T_int, T_ext));

        let bc_int = SurfaceBC::new_interior(8.0, T_int);
        let bc_ext = SurfaceBC::new_exterior(25.0, T_ext);
        fd_solver.step(3600.0, &bc_int, &bc_ext);
        fd_fluxes.push(fd_solver.surface_flux());
    }

    // Compare final flux (should be within 10%)
    let ctf_final = ctf_fluxes[23];
    let fd_final = fd_fluxes[23];

    let diff_percent = 100.0 * (ctf_final - fd_final).abs() / ctf_final;
    assert!(
        diff_percent < 10.0,
        "CTF and FD should agree within 10% (got {}% difference)",
        diff_percent
    );
}

/// Compare 5R1C vs CTF for steady-state
#[test]
fn test_five_r1c_vs_ctf_steady_state() {
    let layers = vec![
        CTFMaterial::new("Concrete", 0.200, 1.4, 2300.0, 880.0),
    ];

    // 5R1C calculation

    for _ in 0..100 {
        ctf_solver.step(20.0, 0.0);
    }
    let ctf_flux = ctf_solver.step(20.0, 0.0);

    // Compare (should be within 5%)
    let diff_percent = 100.0 * (r1c_flux - ctf_flux).abs() / r1c_flux;
    assert!(
        diff_percent < 5.0,
        "5R1C and CTF should agree within 5% at steady-state"
    );
}
